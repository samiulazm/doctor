// var BASE_URL declaration
var BASE_URL = '<?php echo base_url(); ?>';

// Mode toggle (from UI-SPEC Interaction Contracts: Patient Search)
$('input[name="searchMode"]').on('change', function () {
    var mode = $(this).val();
    $('#searchPatientId').toggle(mode === 'id');
    $('#searchDate').toggle(mode === 'date');
});

// Search click handler
$('#btnSearchPatient').on('click', function () {
    var mode = $('input[name="searchMode"]:checked').val();
    var param = mode === 'id' ? { mode: 'id', id: $('#searchPatientId').val() }
                              : { mode: 'date', date: $('#searchDate').val() };
    $.getJSON(BASE_URL + 'doctor_chamber/search_json', param, function (r) {
        if (!r.patients || r.patients.length === 0) {
            $('#searchResultsList').html('<p class="text-muted small">No patients found for this search.</p>');
            return;
        }
        var h = '<p class="small text-muted mb-2">Select a patient to load their history:</p><ul class="list-unstyled mb-0">';
        r.patients.forEach(function (p) {
            h += '<li class="mb-1"><button type="button" class="btn btn-sm btn-outline-primary patient-pick w-100 text-left" data-pid="' + parseInt(p.id, 10) + '">' + $('<div>').text(p.name + ' — ' + p.phone).html() + '</button></li>';
        });
        h += '</ul>';
        $('#searchResultsList').html(h);
    }).fail(function() {
        $('#searchResultsList').html('<p class="text-muted small">Unable to search. Please try again.</p>');
    });
});
$(document).on('click', '.patient-pick', function () {
    var pid = parseInt($(this).data('pid'), 10);
    window.location.href = BASE_URL + 'doctor_chamber/consultation_room?patient=' + pid;
});

// Debounced drug search (280ms) — from UI-SPEC Interaction Contracts
var drugTimer = null;
$('#drugq').on('keyup', function () {
    var t = $(this).val().trim();
    clearTimeout(drugTimer);
    if (t.length < 2) { $('#drugout').empty(); return; }
    drugTimer = setTimeout(function () {
        $.getJSON(BASE_URL + 'doctor_chamber/drug_search_json', { term: t }, function (rows) {
            var h = '';
            (rows || []).forEach(function (x) {
                h += '<li><button type="button" class="btn btn-link btn-sm p-0 text-left drug-pick">' + $('<div>').text(x.label || '').html() + '</button></li>';
            });
            $('#drugout').html(h);
        });
    }, 280);
});
$(document).on('click', '.drug-pick', function () {
    var name = $(this).text().trim();
    var ta = $('#rxScratch');
    ta.val((ta.val() ? ta.val() + '\n' : '') + name);
    $('#drugout').empty();
    $('#drugq').val('');
});

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Doctor browser → search_json | Authenticated GET; search results rendered as button text, not innerHTML |
| Doctor browser → rxFrame iframe | Doctor-initiated submit; prescription module handles its own CSRF |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-05-06-01 | Tampering / XSS | Search results rendered to #searchResultsList | mitigate | Patient name rendered via `$('<div>').text(p.name + ... ).html()` — jQuery text() escapes all special characters before .html() |
| T-05-06-02 | Tampering / XSS | Drug pick rendered to #drugout | mitigate | Same `$('<div>').text(x.label).html()` pattern |
| T-05-06-03 | Tampering / XSS | Vitals rendered to #vitalsBox | mitigate | Vitals values (bp_sys, bp_dia, pulse, weight_kg) are numeric strings from authenticated endpoint; no user-controlled string input reaches the DOM |
| T-05-06-04 | Tampering | Iframe form submit to prescription/addPrescriptionView | accept | Prescription module already has its own CSRF; doctor is authenticated; submit is doctor-initiated |
| T-05-06-05 | Elevation of Privilege | data-pid integer in patient-pick button could be manipulated by doctor | accept | Clicking a manipulated patient-pick would reload consultation_room?patient=N; server-side hospital_id scope in consultation_room() prevents cross-tenant load |
</threat_model>
