```php
public function chart_data_json()
{
    $doc = $this->currentDoctor();
    $labels   = array();
    $revenue  = array();
    $patients = array();

    for ($i = 5; $i >= 0; $i--) {
        $month_num = (int) date('n') - $i;
        $year_num  = (int) date('Y');
        // Adjust for month/year wrap-around
        while ($month_num <= 0) {
            $month_num += 12;
            $year_num  -= 1;
        }
        $start = mktime(0, 0, 0, $month_num, 1, $year_num);
        $end   = mktime(0, 0, 0, $month_num + 1, 1, $year_num);

        $labels[] = date('M', $start);

        // Revenue SUM for this month — payment.date is Unix timestamp INT
        $this->db->select_sum('doctor_amount');
        $this->db->where('hospital_id', $doc->hospital_id);
        $this->db->where('doctor', $doc->id);
        $this->db->where('date >=', $start);
        $this->db->where('date <', $end);
        $rev_row    = $this->db->get('payment')->row();
        $revenue[]  = ($rev_row && $rev_row->doctor_amount !== null) ? (float) $rev_row->doctor_amount : 0.0;

        // Appointment COUNT for this month — appointment.date is also Unix timestamp
        $this->db->where('hospital_id', $doc->hospital_id);
        $this->db->where('doctor', $doc->id);
        $this->db->where('date >=', $start);
        $this->db->where('date <', $end);
        $patients[] = (int) $this->db->count_all_results('appointment');
    }

    $this->output->set_content_type('application/json')
                 ->set_output(json_encode(array(
                     'labels'   => $labels,
                     'revenue'  => $revenue,
                     'patients' => $patients,
                 )));
}
```

Note: The month-wrap logic (`while ($month_num <= 0)`) handles the case where current month is January (month 1) and $i = 5 yields month -4. This is safer than PHP's own mktime month wrap, which works but is less readable.
