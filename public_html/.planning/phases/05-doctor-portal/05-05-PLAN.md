```php
public function search_json()
{
    $doc  = $this->currentDoctor();
    $mode = $this->input->get('mode');   // 'id' or 'date'
    $out  = array('patients' => array());

    if ($mode === 'id') {
        $patient_id = (int) $this->input->get('id');
        if ($patient_id < 1) {
            $this->output->set_content_type('application/json')->set_output(json_encode($out));
            return;
        }
        $this->db->where('hospital_id', $doc->hospital_id);
        $this->db->where('id', $patient_id);
        $this->db->limit(1);
        $rows = $this->db->get('patient_list')->result();
        foreach ($rows as $r) {
            $out['patients'][] = array(
                'id'    => (int) $r->id,
                'name'  => (string) $r->name,
                'phone' => (string) ($r->phone ?? ''),
            );
        }
    } elseif ($mode === 'date') {
        $date_str = $this->input->get('date');   // YYYY-MM-DD from date input
        $ts_start = $date_str ? strtotime($date_str) : false;
        if (!$ts_start) {
            $this->output->set_content_type('application/json')->set_output(json_encode($out));
            return;
        }
        $ts_end = $ts_start + 86400;
        // Find distinct patients seen by this doctor on that date via appointment table
        $this->db->select('p.id, p.name, p.phone');
        $this->db->from('appointment a');
        $this->db->join('patient_list p', 'p.id = a.patient', 'inner');
        $this->db->where('a.hospital_id', $doc->hospital_id);
        $this->db->where('a.doctor', $doc->id);
        $this->db->where('a.date >=', $ts_start);
        $this->db->where('a.date <', $ts_end);
        $this->db->group_by('p.id');
        $this->db->limit(50);
        $rows = $this->db->get()->result();
        foreach ($rows as $r) {
            $out['patients'][] = array(
                'id'    => (int) $r->id,
                'name'  => (string) $r->name,
                'phone' => (string) ($r->phone ?? ''),
            );
        }
    }

    $this->output->set_content_type('application/json')->set_output(json_encode($out));
}
```

Note: Uses `patient_list` (the view/table used throughout the codebase for patient lookups — same as used in `patient_model->getPatientById()`). Scoped to $doc->hospital_id. Integer cast on ID input prevents injection. Date input validated via strtotime() returning false on invalid strings.
