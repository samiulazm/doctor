```php
public function mark_queue_fee_paid_ajax()
{
    if ($this->input->method() !== 'post') {
        $this->output->set_status_header(405)->set_content_type('application/json')
             ->set_output(json_encode(array('success' => false, 'message' => 'Method not allowed')));
        return;
    }
    $id     = (int) $this->input->post('queue_id');
    $hid    = $this->session->userdata('hospital_id');
    $row    = $this->queue_model->getRow($id, $hid);
    $amount = (float) $this->input->post('amount');
    if ($amount <= 0) {
        $this->output->set_content_type('application/json')
             ->set_output(json_encode(array('success' => false, 'message' => 'Invalid amount')));
        return;
    }
    $pay_status  = in_array($this->input->post('pay_status'), array('paid', 'due'), true)
        ? $this->input->post('pay_status') : 'paid';
    $fee_type    = in_array($this->input->post('fee_type'), array('consultation', 'procedure'), true)
        ? $this->input->post('fee_type') : 'consultation';
    $pay_method  = $this->input->post('pay_method') ?: 'cash';
    $remarks     = $this->input->post('remarks') ?: '';
    $patient_id  = $row ? (int) $row->patient_id : 0;
    $doctor_id   = $row ? (int) $row->doctor_id  : 0;
    $payment_data = array(
        'hospital_id'    => $hid,
        'patient_id'     => $patient_id,
        'doctor_id'      => $doctor_id,
        'amount'         => $amount,
        'fee_type'       => $fee_type,
        'pay_method'     => $pay_method,
        'status'         => $pay_status,
        'remarks'        => $remarks,
        'queue_id'       => $id,
        'created_at'     => date('Y-m-d H:i:s'),
    );
    $this->finance_model->insertPayment($payment_data);
    $this->output->set_content_type('application/json')->set_output(json_encode(array(
        'success'   => true,
        'message'   => 'Fee recorded.',
        'csrf_hash' => $this->security->get_csrf_hash(),
    )));
}
```
