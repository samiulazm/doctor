```php
public function getPrescriptionForPortal($id)
{
    $this->db->select('prescription.*, doctor.name AS doctor_name, doctor.degree AS doctor_degree');
    $this->db->join('doctor', 'doctor.id = prescription.doctor_id', 'left');
    $this->db->where('prescription.id', (int) $id);
    return $this->db->get('prescription')->row();
}
```

```php
public function prescription($id = 0)
{
    $id    = (int) $id;
    $phone = $this->session->userdata('portal_verified_phone');
    if (!$phone || !$id) {
        show_404();
    }

    $hospital_id = (int) $this->session->userdata('portal_verified_hospital');
    $rx = $this->portal_model->getPrescriptionForPortal($id);
    if (!$rx) {
        show_404();
    }

    // Ownership check: prescription.patient must match patient looked up by phone
    $patient = $this->portal_model->findPatientByPhone($hospital_id, $phone);
    if (!$patient || (int) $rx->patient !== (int) $patient->id) {
        show_error('Access denied', 403);
    }

    // Parse medicine string
    $medicines = array();
    if (!empty($rx->medicine)) {
        $entries = explode('###', $rx->medicine);
        foreach ($entries as $entry) {
            $fields = explode('***', $entry);
            if (count($fields) >= 4) {
                $medicines[] = array(
                    'name'        => htmlspecialchars(trim($fields[0]), ENT_QUOTES),
                    'dose'        => htmlspecialchars(isset($fields[1]) ? trim($fields[1]) : '', ENT_QUOTES),
                    'frequency'   => htmlspecialchars(isset($fields[2]) ? trim($fields[2]) : '', ENT_QUOTES),
                    'days'        => htmlspecialchars(isset($fields[3]) ? trim($fields[3]) : '', ENT_QUOTES),
                    'instruction' => htmlspecialchars(isset($fields[4]) ? trim($fields[4]) : '', ENT_QUOTES),
                );
            }
        }
    }

    // Look up medicine names from medicine table if fields[0] is a numeric ID
    // (fields[0] is medicine_id when prescription written via e-pad; may already be name string in older prescriptions)
    foreach ($medicines as &$med) {
        if (is_numeric($med['name'])) {
            $med_row = $this->db->get_where('medicine', array('id' => (int) $med['name']), 1)->row();
            if ($med_row) {
                $med['name'] = htmlspecialchars($med_row->name ?? ($med_row->brand_name ?? $med['name']), ENT_QUOTES);
            }
        }
    }
    unset($med);

    $data = array(
        'rx'        => $rx,
        'medicines' => $medicines,
        'patient'   => $patient,
        'rx_id'     => $id,
    );

    $this->load->view(
        'portal/layout_public',
        array('content' => $this->load->view('portal/rx_detail', $data, true))
    );
}
```

```php
public function prescription_pdf($id = 0)
{
    $id    = (int) $id;
    $phone = $this->session->userdata('portal_verified_phone');
    if (!$phone || !$id) {
        show_404();
    }

    $hospital_id = (int) $this->session->userdata('portal_verified_hospital');
    $rx = $this->portal_model->getPrescriptionForPortal($id);
    if (!$rx) {
        show_404();
    }

    $patient = $this->portal_model->findPatientByPhone($hospital_id, $phone);
    if (!$patient || (int) $rx->patient !== (int) $patient->id) {
        show_error('Access denied', 403);
    }

    // Parse medicine (same logic as prescription())
    $medicines = array();
    if (!empty($rx->medicine)) {
        $entries = explode('###', $rx->medicine);
        foreach ($entries as $entry) {
            $fields = explode('***', $entry);
            if (count($fields) >= 4) {
                $medicines[] = array(
                    'name'        => trim($fields[0]),
                    'dose'        => isset($fields[1]) ? trim($fields[1]) : '',
                    'frequency'   => isset($fields[2]) ? trim($fields[2]) : '',
                    'days'        => isset($fields[3]) ? trim($fields[3]) : '',
                    'instruction' => isset($fields[4]) ? trim($fields[4]) : '',
                );
            }
        }
    }
    foreach ($medicines as &$med) {
        if (is_numeric($med['name'])) {
            $med_row = $this->db->get_where('medicine', array('id' => (int) $med['name']), 1)->row();
            if ($med_row) {
                $med['name'] = $med_row->name ?? ($med_row->brand_name ?? $med['name']);
            }
        }
    }
    unset($med);

    $data = array(
        'rx'        => $rx,
        'medicines' => $medicines,
        'patient'   => $patient,
        'rx_id'     => $id,
    );

    $html = $this->load->view('portal/prescription_pdf_template', $data, true);

    // Ensure mPDF temp directory exists
    $tmp_dir = APPPATH . '../files/mpdf-tmp';
    if (!is_dir($tmp_dir)) {
        mkdir($tmp_dir, 0755, true);
    }

    $mpdf = new \Mpdf\Mpdf(array('format' => 'A4', 'tempDir' => $tmp_dir));
    $mpdf->setAutoTopMargin    = 'stretch';
    $mpdf->setAutoBottomMargin = 'stretch';
    $mpdf->WriteHTML($html);
    $mpdf->Output('prescription-' . $id . '.pdf', 'D');
    exit;
}
```
