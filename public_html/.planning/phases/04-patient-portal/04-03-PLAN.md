```php
public function slots_json()
{
    $chamber_id = (int) $this->input->get('chamber_id');
    $date       = $this->input->get('date');

    if (!$chamber_id || !$date || !preg_match('/^\d{4}-\d{2}-\d{2}$/', $date)) {
        $this->output->set_content_type('application/json')
            ->set_output(json_encode(array('ok' => false)));
        return;
    }

    // Validate chamber exists and belongs to an enabled hospital
    $chamber = $this->db->get_where('doctor_chamber', array('id' => $chamber_id, 'is_active' => 1), 1)->row();
    if (!$chamber || !chamber_practice_enabled_for_hospital($this, (int) $chamber->hospital_id)) {
        $this->output->set_content_type('application/json')
            ->set_output(json_encode(array('ok' => false)));
        return;
    }

    // Check exception table (is_closed override for specific date)
    $exception = $this->db->get_where('doctor_schedule_exception', array(
        'chamber_id'     => $chamber_id,
        'exception_date' => $date,
        'is_closed'      => 1,
    ), 1)->row();
    if ($exception) {
        $this->output->set_content_type('application/json')
            ->set_output(json_encode(array(
                'ok'  => false,
                'msg' => 'This chamber is closed on the selected date. Please choose another date.',
            )));
        return;
    }

    // Get weekly hours for day of week (0=Sunday)
    $hours   = json_decode($chamber->weekly_hours_json, true);
    $day_num = (int) date('w', strtotime($date)); // 0=Sun...6=Sat
    $day_key = (string) $day_num;

    if (empty($hours[$day_key]) || empty($hours[$day_key]['s_time']) || empty($hours[$day_key]['e_time'])) {
        $this->output->set_content_type('application/json')
            ->set_output(json_encode(array(
                'ok'  => false,
                'msg' => 'No regular hours are available for this chamber on the selected day.',
            )));
        return;
    }

    $s_time = strtotime($date . ' ' . $hours[$day_key]['s_time']);
    $e_time = strtotime($date . ' ' . $hours[$day_key]['e_time']);
    $slots  = array();
    for ($t = $s_time; $t < $e_time; $t += 1800) {
        $slots[] = array(
            'time'      => date('H:i', $t),
            'label'     => date('g:i A', $t),
            'available' => true,
        );
    }

    $this->output->set_content_type('application/json')
        ->set_output(json_encode(array('ok' => true, 'slots' => $slots)));
}
```

Do not use raw SQL queries. All DB access through CI Query Builder (`$this->db->`).

```javascript
function loadSlots() {
    var chamberId = $('#chamberSelect').val();
    var bookingDate = $('#bookingDate').val();
    if (!chamberId || !bookingDate) return;
    $('#slotGrid').html('<p class="text-muted small">Loading slots...</p>');
    $.getJSON('<?php echo site_url('portal/slots_json'); ?>', {
        chamber_id: chamberId, date: bookingDate
    }, function(r) {
        if (!r.ok) {
            $('#slotGrid').html('<p class="text-danger small">' + (r.msg || 'No slots available.') + '</p>');
            return;
        }
        var html = '';
        $.each(r.slots, function(i, s) {
            var cls = s.available ? 'chamber-slot-pill' : 'chamber-slot-pill full';
            html += '<button type="button" class="' + cls + '" data-time="' + s.time + '">' + s.label + '</button>';
        });
        $('#slotGrid').html(html);
    });
}
$('#chamberSelect, #bookingDate').on('change', loadSlots);
$(document).on('click', '.chamber-slot-pill:not(.full)', function() {
    $('.chamber-slot-pill').removeClass('selected');
    $(this).addClass('selected');
    $('#selectedSlot').val($(this).data('time'));
});
```
