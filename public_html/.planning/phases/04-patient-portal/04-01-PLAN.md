**Step 1 — Portal_model.php additions (append 3 methods before the closing `}`):**

```php
public function getUpcomingAppointment($hospital_id, $patient_id)
{
    $this->db->where('hospital_id', (int) $hospital_id);
    $this->db->where('patient_id', (int) $patient_id);
    $this->db->where_in('status', array('pending', 'confirmed', 'waiting'));
    $this->db->where('appointment_date >=', date('Y-m-d'));
    $this->db->order_by('appointment_date', 'ASC');
    $this->db->limit(1);
    return $this->db->get('appointment_list')->row();
}

public function getPrescriptionsForPatient($hospital_id, $patient_id)
{
    $this->db->where('hospital_id', (int) $hospital_id);
    $this->db->where('patient', (int) $patient_id);
    $this->db->order_by('created_at', 'DESC');
    $this->db->limit(10);
    return $this->db->get('prescription')->result();
}

public function getLabReportsForPatient($hospital_id, $patient_id)
{
    // Run DESCRIBE lab_report at execution time to confirm the patient FK column name.
    // Likely 'patient_id' — fall back to checking lab_report_list if needed.
    $this->db->where('hospital_id', (int) $hospital_id);
    $this->db->where('patient_id', (int) $patient_id);
    $this->db->order_by('id', 'DESC');
    $this->db->limit(10);
    return $this->db->get('lab_report')->result();
}
```
